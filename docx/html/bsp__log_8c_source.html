74err_t BSPLogInit(void) {
75#if BSP_LOG_ENABLED
76 err_t result = OK;
77
78 if (!BSPLogInThreadMode() ||
79 xTaskGetSchedulerState() != taskSCHEDULER_NOT_STARTED) {
80 return BSPLogRecordResult(STATE_ERR);
81 }
82 if (log_initialized) {
83 return BSPLogRecordResult(OK);
84 }
85
86 SEGGER_RTT_Init();
87
88 if (SEGGER_RTT_SetFlagsUpBuffer(BSP_LOG_CHANNEL_TERMINAL,
89 SEGGER_RTT_MODE_NO_BLOCK_SKIP) < 0) {
90 return BSPLogRecordResult(INIT_ERR);
91 }
92 result = (SEGGER_RTT_ConfigUpBuffer(BSP_LOG_CHANNEL_DATA, "JScope_t4f4f4f4",
93 data_buffer, sizeof(data_buffer),
94 SEGGER_RTT_MODE_NO_BLOCK_SKIP) < 0)
95 ? INIT_ERR
96 : OK;
97 if (result == OK) {
98 result = (SEGGER_RTT_ConfigUpBuffer(
99 BSP_LOG_CHANNEL_RESERVED, "Reserved", reserved_buffer,
100 sizeof(reserved_buffer), SEGGER_RTT_MODE_NO_BLOCK_SKIP) < 0)
101 ? INIT_ERR
102 : OK;
103 }
104 if (result == OK) {
105 log_mutex = xSemaphoreCreateMutexStatic(&log_mutex_storage);
106 if (log_mutex == NULL) {
107 return BSPLogRecordResult(INIT_ERR);
108 }
109 /* 所有通道配置成功后，原子发布默认等级和可用状态。 */
110 atomic_store_explicit(&current_level, BSP_LOG_DEFAULT_LEVEL,
111 memory_order_release);
112 log_initialized = true;
113 }
114 return BSPLogRecordResult(result);
115#else
116 return BSPLogRecordResult(NOT_SUPPORT);
117#endif
118}

120err_t BSPLogWrite(bsp_log_level_t level, const char *tag, const char *format,
121 ...) {
122#if BSP_LOG_ENABLED
123 const bsp_log_event_t event = {
124 .level = level,
125 .tag = tag,
126 .format = format,
127 };
128 va_list args;
129 err_t result;
130 err_t unlock_result;
131
132 if (BSPLogValidateLevel(level) != OK) {
133 return BSPLogRecordResult(ARG_ERR);
134 }
135 if (tag == NULL || format == NULL) {
136 return BSPLogRecordResult(PTR_NULL);
137 }
138 if (!log_initialized) {
139 return BSPLogRecordResult(INIT_ERR);
140 }
141 /* 等级过滤不执行格式化或 RTT 输出，也不修改已保存的错误状态。
142 * PENDING 表示本条记录被丢弃，不存在排队或稍后补发。 */
143 if (level < (bsp_log_level_t)atomic_load_explicit(&current_level,
144 memory_order_acquire)) {
145 return PENDING;
146 }
147 if (!BSPLogContextAllowsWrite()) {
148 return BSPLogRecordResult(STATE_ERR);
149 }
150
151 result = BSPLogLock();
152 if (result != OK) {
153 return BSPLogRecordResult(result);
154 }
155
156 /* 正文可变参数只建立一次，所有输出结果都先清理参数列表，再释放锁。 */
157 va_start(args, format);
158 result = BSPLogTerminalCallback(&event, &args);
159 va_end(args);
160 unlock_result = BSPLogUnlock();
161 return BSPLogRecordResult(unlock_result != OK ? unlock_result : result);
162#else
163 /* 日志关闭时，直接调用接口仍返回明确状态；LOG_* 宏则完全不求值参数。 */
164 RM_UNUSED(level);
165 RM_UNUSED(tag);
166 RM_UNUSED(format);
167 return BSPLogRecordResult(NOT_SUPPORT);
168#endif
169}

215err_t BSPLogPollLevelCommand(void) {
216#if BSP_LOG_ENABLED
217 int byte;
218 err_t result = PENDING;
219
220 if (!BSPLogContextAllowsWrite()) {
221 return BSPLogRecordResult(STATE_ERR);
222 }
223 if (!log_initialized) {
224 return BSPLogRecordResult(INIT_ERR);
225 }
226
227 /* GetKey 固定读取下行通道 0；无数据时返回 -1，不等待主机。 */
228 while ((byte = SEGGER_RTT_GetKey()) >= 0) {
229 if (byte == '\r' || byte == '\n') {
230 if (level_command_size == 0u) {
231 continue;
232 }
233 result = level_command_size < sizeof(level_command_buffer)
234 ? BSPLogApplyLevelCommand()
235 : BSPLogRecordResult(ARG_ERR);
236 level_command_size = 0u;
237 } else if (level_command_size < sizeof(level_command_buffer)) {
238 level_command_buffer[level_command_size++] = (char)byte;
239 }
240 }
241
242 return result;
243#else
244 return BSPLogRecordResult(NOT_SUPPORT);
245#endif
246}

263int Float2StrN(char *str, size_t size, float va) {
267 static const double scaled_integer_limit = 0x1p63;
268 double scaled_rounded;
269 uint64_t
270 scaled_magnitude;
271 uint64_t integer_part;
272 uint32_t
273 fraction_part;
274 const char *sign;
275 int result;
276
277 if (str == NULL || size == 0u) {
278 return -1;
279 }
280 str[0] = '\0';
281 if (!isfinite(va)) {
282 return -1;
283 }
284
285 /* 先缩放取整（中点远离零），再检查转换范围。 */
286 scaled_rounded = round((double)va * FRACTION_SCALE);
287 if (scaled_rounded < -scaled_integer_limit ||
288 scaled_rounded >= scaled_integer_limit) {
289 return -1;
290 }
291
292 scaled_magnitude = (uint64_t)fabs(scaled_rounded);
293 integer_part = scaled_magnitude / FRACTION_SCALE;
294 fraction_part = (uint32_t)(scaled_magnitude % FRACTION_SCALE);
295 sign = scaled_rounded < 0.0 ? "-" : ""; /* 取整为零时统一输出正零。 */
296
297 /* 分组输出 32 位整数，兼容不支持 long long 的 newlib-nano。 */
298 if (integer_part < INTEGER_GROUP_BASE) {
299 result = snprintf(str, size, "%s%" PRIu32 ".%0*" PRIu32, sign,
300 (uint32_t)integer_part, FRACTION_DIGITS, fraction_part);
301 } else {
302 result = snprintf(str, size, "%s%" PRIu32 "%0*" PRIu32 ".%0*" PRIu32, sign,
303 (uint32_t)(integer_part / INTEGER_GROUP_BASE),
304 INTEGER_GROUP_DIGITS,
305 (uint32_t)(integer_part % INTEGER_GROUP_BASE),
306 FRACTION_DIGITS, fraction_part);
307 }
308 if (result < 0) {
309 str[0] = '\0';
310 return -1;
311 }
312 return result;
313}

351static bool BSPLogContextAllowsWrite(void) {
352 BaseType_t scheduler_state;
353
354 if (!BSPLogInThreadMode()) {
355 return false;
356 }
357 /* 只查询一次调度器状态，后续判断使用同一次结果。NOT_STARTED
358 * 使用启动阶段状态，RUNNING 允许任务输出；SUSPENDED 拒绝操作日志互斥锁。 */
359 scheduler_state = xTaskGetSchedulerState();
360
361 /* 调度器状态和两组中断屏蔽条件必须同时满足。 */
362 return (scheduler_state == taskSCHEDULER_NOT_STARTED ||
363 scheduler_state == taskSCHEDULER_RUNNING) &&
364 /* 要求 PRIMASK 为 0，避免在可配置优先级异常被全局屏蔽期间格式化。
365 * NMI 和 HardFault 不受 PRIMASK 屏蔽。 */
366 (__get_PRIMASK() == BSP_LOG_PRIMASK_UNMASKED) &&
367 /* 还要求 BASEPRI 为 0；即使 PRIMASK 为 0，FreeRTOS 临界区仍可能
368 * 通过 BASEPRI 屏蔽较低优先级异常。 */
369 (__get_BASEPRI() == BSP_LOG_BASEPRI_UNMASKED);
370}

436static bool BSPLogMatchIgnoreCase(const char *value, const char *pattern,
437 bool exact_match) {
438 while (*pattern != '\0' && *value != '\0') {
439 char value_char = *value++;
440 char pattern_char = *pattern++;
441 if (value_char >= 'a' && value_char <= 'z') {
442 value_char = (char)(value_char - ('a' - 'A'));
443 }
444 if (pattern_char >= 'a' && pattern_char <= 'z') {
445 pattern_char = (char)(pattern_char - ('a' - 'A'));
446 }
447 if (value_char != pattern_char) {
448 return false;
449 }
450 }
451
452 if (*pattern != '\0') {
453 return false;
454 }
455 return !exact_match || *value == '\0';
456}

465static char *BSPLogTrimCommand(char *command) {
466 char *first = command;

490static err_t BSPLogParseLevelCommand(const char *command,
491 bsp_log_level_t *level) {
492 static const char *const names[] = {"TRACE", "DEBUG", "INFO",
493 "WARN", "ERROR", "FATAL"};
494 const char *value;
495 size_t index;
496
497 /* 判断命令格式是否符合log level */
498 if (!BSPLogMatchIgnoreCase(command, "log level", false) ||
499 (command[9] != ' ' && command[9] != '\t')) {
500 return ARG_ERR;
501 }
502 value = command + 9;
503 while (*value == ' ' || *value == '\t') {
504 ++value;
505 }
506
507 for (index = 0u; index < ARRAY_LEN(names); ++index) {
508 if (BSPLogMatchIgnoreCase(value, names[index], true)) {
509 *level = (bsp_log_level_t)index;
510 return OK;
511 }
512 }
513
514 return ARG_ERR;
515}
